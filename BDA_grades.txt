
# MapReduce Program to Find Grades of Students

# Grade Calculation Function
def calculate_grade(marks):

    if marks >= 90:
        return "A"

    elif marks >= 75:
        return "B"

    elif marks >= 50:
        return "C"

    else:
        return "Fail"


#Mapper Function - Reads each line from the file.
#Splits student name and marks.
#Calculates grade using calculate_grade().
#Creates key-value pairs:

def mapper(lines):

    mapped = []

    for line in lines:

        # Split student name and marks
        data = line.strip().split()

        name = data[0]
        marks = int(data[1])

        grade = calculate_grade(marks)

        mapped.append((name, grade))

    return mapped


#reducer function - Stores final student grades in a dictionary

def reducer(mapped_data):
    result = {}

    for name, grade in mapped_data:
        result[name] = grade
    return result



# Read Input File

with open("students.txt", "r") as file:
    lines = file.readlines()

# Map phase - Generates (student, grade) pairs.
mapped_data = mapper(lines)

print("Mapper Output:")
print(mapped_data)


# Reduce Phase - Produces final grade results.
reduced_data = reducer(mapped_data)


print("\nStudent Grades:")

for name, grade in reduced_data.items():
    print(name, ":", grade)
