
# MapReduce Program to Find Frequency of a Given Word


# Mapper Function - Reads each line from the file.
#Splits lines into words using split().
#Compares every word with the target word.
#If matched, emits:
def mapper(lines, target_word):
    mapped = []
    for line in lines:
        words = line.strip().split()
        for word in words:

            # Convert to lowercase for accurate comparison
            if word.lower() == target_word.lower():

                mapped.append((target_word, 1))

    return mapped


# Reducer Function - Adds all values from mapper output.
# Calculates total frequency of the target word.
def reducer(mapped_data):
    count = 0
    for word, value in mapped_data:
        count += value
    return count



# Read Input File
with open("input.txt", "r") as file:
    lines = file.readlines()


# Target Word
target_word = input("Enter word to search: ")


# Map Phase - Generates intermediate key-value pairs for matching words.

mapped_data = mapper(lines, target_word)
print("\nMapper Output:")
print(mapped_data)


# Reduce Phase - Counts total occurrences of the target word.

frequency = reducer(mapped_data)


print("\nWord Frequency:")
print(target_word, ":", frequency)
