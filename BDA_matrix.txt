
# Matrix Multiplication using MapReduce

# Matrix A
A = [
    [1, 2],
    [3, 4]
]

# Matrix B
B = [
    [5, 6],
    [7, 8]
]


# Mapper Function - Performs multiplication of matrix elements
# Generates intermediate key-value pairs:

def mapper(A, B):

    mapped = []

    rows_A = len(A)
    cols_A = len(A[0])

    rows_B = len(B)
    cols_B = len(B[0])

    # Matrix Multiplication Mapping 
    for i in range(rows_A):
        for j in range(cols_B):
            for k in range(cols_A):
                value = A[i][k] * B[k][j] # Calculates multiplication of corresponding elements
                mapped.append(((i, j), value)) #(i, j) → position in result matrix
                                               #value → partial multiplication result
    return mapped



# Reducer Function - Groups same matrix positions
# Adds partial values to get final matrix element

def reducer(mapped_data):
    result = {}
    for key, value in mapped_data:
        if key in result:
            result[key] += value
        else:
            result[key] = value

    return result



# Map Phase -Produces intermediate multiplication results

mapped_data = mapper(A, B)

print("Mapper Output:")

for item in mapped_data:
    print(item)


# Reduce Phase -Calculates final values of the result matrix

reduced_data = reducer(mapped_data)


# Construct Final Matrix

rows = len(A)
cols = len(B[0])

result_matrix = [
    [0 for j in range(cols)]
    for i in range(rows)
]

for (i, j), value in reduced_data.items():

    result_matrix[i][j] = value   #Stores reduced values into the final matrix


print("\nResultant Matrix:")

for row in result_matrix:
    print(row)
