# ---------------------------------------------------------
# MapReduce Program for Titanic Data Analysis
# ---------------------------------------------------------
# Tasks:
# 1. Find average age of dead males
# 2. Find number of dead females in each class
# ---------------------------------------------------------

import csv

# ---------------------------------------------------------
# Mapper Function
# ---------------------------------------------------------
#Processes each passenger record:
def mapper(file_name):

    male_age = []

    female_class = []

    with open(file_name, "r") as file:

        reader = csv.DictReader(file) #Reads Titanic dataset (titanic.csv)
                                     #Converts each row into a dictionary format

        for row in reader:

            # Convert values
            survived = int(row['survived'])

            sex = row['sex']

            pclass = row['pclass']

            age = row['age']

            # -------------------------------------------------
            # Dead Male Average Age -Stores their ages for averaging
            # -------------------------------------------------

            if survived == 0 and sex == "male":

                if age != "":

                    male_age.append(("male_age", float(age)))

            # -------------------------------------------------
            # Dead Females in Each Class -Stores (class, 1) for counting
            # -------------------------------------------------

            if survived == 0 and sex == "female":

                female_class.append((pclass, 1))

    return male_age, female_class


# ---------------------------------------------------------
# Reducer Function
# ---------------------------------------------------------

def reducer(male_age, female_class):

    # -----------------------------------------------------
    # Average Age of Dead Males
    # -----------------------------------------------------

    total_age = 0
    count = 0

    for key, age in male_age:

        total_age += age
        count += 1

    average_age = total_age / count

    # -----------------------------------------------------
    # Dead Females Count by Class
    # -----------------------------------------------------

    female_count = {}

    for pclass, value in female_class:

        if pclass in female_count:
            female_count[pclass] += value #Groups and counts dead females by passenger class

        else:
            female_count[pclass] = value

    return average_age, female_count
#male_age → list of ages of dead males, female_class → list of dead females per class

# ---------------------------------------------------------
# Main Program
# ---------------------------------------------------------

male_age, female_class = mapper("titanic.csv")

average_age, female_count = reducer(
    male_age,
    female_class
)

# ---------------------------------------------------------
# Output -Computes average age of dead males, females
# ---------------------------------------------------------

print("Average Age of Dead Males:")
print(round(average_age, 2))

print("\nDead Females in Each Class:")

for pclass, count in female_count.items():

    print("Class", pclass, ":", count)
